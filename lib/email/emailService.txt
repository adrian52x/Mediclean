import nodemailer from 'nodemailer';

interface OrderDetails {
  orderId: string;
  timestamp: string;
  customer: {
    name: string;
    phone: string;
    email: string;
  };
  delivery: {
    method: string;
    address: any;
    notes: string;
  };
  items: Array<{
    productId: string;
    title: string;
    volume?: string;
    price: number;
    quantity: number;
    total: number;
  }>;
  summary: {
    totalItems: number;
    totalPrice: number;
    deliveryFee: number;
    finalTotal: number;
    currency: string;
  };
}

export const sendOrderConfirmationEmail = async (orderDetails: OrderDetails) => {
  try {
    console.log('📧 Starting email service...');
    console.log('📧 Gmail Email:', process.env.GMAIL_EMAIL ? 'Configured' : 'Missing');
    console.log('📧 Gmail Password:', process.env.GMAIL_APP_PASSWORD ? 'Configured' : 'Missing');

    // Create transporter using explicit Gmail SMTP settings
    const transporter = nodemailer.createTransport({
      host: 'smtp.gmail.com',
      port: 465,
      secure: true, // true for 465, false for other ports
      auth: {
        user: process.env.GMAIL_EMAIL,
        pass: process.env.GMAIL_APP_PASSWORD,
      },
      tls: {
        rejectUnauthorized: false, // Allow self-signed certificates
        servername: 'smtp.gmail.com'
      }
    });

    console.log('📧 Transporter created, verifying connection...');
    
    // Test the connection
    try {
      await transporter.verify();
      console.log('✅ SMTP connection verified successfully');
    } catch (verifyError) {
      console.error('❌ SMTP verification failed:', verifyError);
      throw verifyError;
    }

    const mailOptions = {
      from: {
        name: 'Mediclean - Confirmare comandă',
        address: process.env.GMAIL_EMAIL!,
      },
      to: orderDetails.customer.email,
      subject: `Confirmare comandă #${orderDetails.orderId} - Mediclean`,
      html: createEmailTemplate(orderDetails),
    };

    console.log('📧 Sending email to:', orderDetails.customer.email);
    const result = await transporter.sendMail(mailOptions);
    console.log('✅ Email sent successfully:', result.messageId);
    return { success: true, messageId: result.messageId };

  } catch (error) {
    console.error('❌ Failed to send email:', error);
    return { success: false, error: error };
  }
};
